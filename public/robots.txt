# ── Content Signals (contentsignals.org / IETF draft-romm-aipref-contentsignals) ──
# Independent of crawl access above: declares how this content may be used.
# search=yes   -> ok to index for search results
# ai-input=yes -> ok to use as grounding/context for AI answers (RAG, citations)
# ai-train=no  -> not ok to use for model training

# blocked by name below, so the signal and the crawl rules agree. Search and retrieval
# crawlers stay allowed so the site can still be found and cited.
